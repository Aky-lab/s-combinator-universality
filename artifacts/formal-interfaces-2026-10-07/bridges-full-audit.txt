  (registers : PureSFormal.PureS.SchedulerControl.Registers program) (bit : Bool) (suffix : List Bool)
  (outerContext fullContext innerContext targetContext : PureSFormal.PureS.Context)
  (parents : List PureSFormal.PureS.ParentFrame) (ticks : Nat)
  (trace :
    PureSFormal.PureS.SchedulerCycle.SelectedResponseTrace program dispatcher seedBits continuation source admissible
      registers bit suffix outerContext fullContext innerContext targetContext parents ticks)
  (notSeen : registers.seen = false) :
  ∃ configurations last,
    PureSFormal.PureS.SchedulerResponseInvariant.ExactMutationChain
        (PureSFormal.PureS.SchedulerControl.machine program dispatcher)
        (PureSFormal.PureS.SchedulerCycle.selectedReturnConfiguration program dispatcher registers bit suffix seedBits
          continuation (PureSFormal.PureS.SchedulerCycle.deletedCarrier bit outerContext innerContext) parents)
        (PureSFormal.PureS.SchedulerInvariant.upConfiguration program dispatcher registers PureSFormal.PureS.omega

            (PureSFormal.PureS.ParentFrame.right
                (PureSFormal.PureS.SchedulerResponse.pendingFunction program dispatcher seedBits continuation) ::
              parents)))
        configurations ∧
      configurations.length =
          1 +
            PureSFormal.PureS.LocalResponse.completedCost program
              (dispatcher.route ((PureSFormal.PureS.SchedulerCycle.scannedRegisters registers bit suffix).phase, bit))
              ((PureSFormal.PureS.SchedulerCycle.scannedRegisters registers bit suffix).phase, bit) ∧
        PureSFormal.PureS.SchedulerTraceAlgebra.LastSample configurations last ∧
          last.cursor.erase =
            PureSFormal.PureS.Cursor.rebuild parents
              (PureSFormal.PureS.LocalResponse.completed seedBits continuation
